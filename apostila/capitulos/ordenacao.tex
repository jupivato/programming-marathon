\section{Bubble sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n^2)$ (melhor, médio e pior caso).

\textbf{Complexidade espacial:} $O(1)$.

%------------------------------------------------------------------------------
\section{Selection sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n^2)$.

\textbf{Complexidade espacial:} $O(1)$.

\incluircpp{algoritmos_avulsos/ordenacao/selection_sort.cpp}

%------------------------------------------------------------------------------
\section{Insertion sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n)$ no melhor caso; $O(n^2)$ no médio e no pior.

\textbf{Complexidade espacial:} $O(1)$.

\incluircpp{algoritmos_avulsos/ordenacao/insertion_sort.cpp}

%------------------------------------------------------------------------------
\section{Shell sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} entre $O(n \log n)$ e $O(n^2)$, conforme a sequência de gaps.

\textbf{Complexidade espacial:} $O(1)$.

\incluircpp{algoritmos_avulsos/ordenacao/shell_sort.cpp}

\section{Merge sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n \log n)$.

\textbf{Complexidade espacial:} $O(n)$ (vetores auxiliares na intercalação).

\incluircpp{algoritmos_avulsos/ordenacao/merge_sort.cpp}

%------------------------------------------------------------------------------
\section{Quick sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n \log n)$ em média; $O(n^2)$ no pior caso.

\textbf{Complexidade espacial:} $O(\log n)$ na pilha de recursão (in-place no vetor).

\incluircpp{algoritmos_avulsos/ordenacao/quick_sort.cpp}

%------------------------------------------------------------------------------
\section{Heap sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n \log n)$.

\textbf{Complexidade espacial:} $O(1)$ além do vetor.

\incluircpp{algoritmos_avulsos/ordenacao/heapsort.cpp}

\section{Counting sort}
%------------------------------------------------------------------------------

\textbf{Complexidade temporal:} $O(n + k)$, com $k$ = faixa de valores (ex.: $0 \ldots k-1$).

\textbf{Complexidade espacial:} $O(k)$.

\incluircpp{algoritmos_avulsos/ordenacao/counting_sort.cpp}
